set(SHADER_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../shaders)
set(GENERATED_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(GLOB SHADER_SOURCES CONFIGURE_DEPENDS ${SHADER_DIRECTORY}/*.glsl)
list(SORT SHADER_SOURCES)
set(EMBEDDED_SHADERS "#pragma once\n// Generated by CMake from src/main/shaders. Do not edit.\n\nnamespace nanquim::shaders {\n")
foreach(SHADER_PATH ${SHADER_SOURCES})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SHADER_PATH})
  get_filename_component(SHADER_NAME ${SHADER_PATH} NAME_WE)
  file(READ ${SHADER_PATH} SHADER_TEXT)
  string(APPEND EMBEDDED_SHADERS "\ninline constexpr const char* ${SHADER_NAME} = R\"nanquim_glsl(${SHADER_TEXT})nanquim_glsl\";\n")
endforeach()
string(APPEND EMBEDDED_SHADERS "\nstruct EmbeddedShader {\n  const char* name;\n  const char* source;\n};\n\ninline constexpr EmbeddedShader kEmbeddedShaders[] = {\n")
foreach(SHADER_PATH ${SHADER_SOURCES})
  get_filename_component(SHADER_NAME ${SHADER_PATH} NAME_WE)
  string(APPEND EMBEDDED_SHADERS "    {\"${SHADER_NAME}\", ${SHADER_NAME}},\n")
endforeach()
string(APPEND EMBEDDED_SHADERS "};\n")
string(APPEND EMBEDDED_SHADERS "\n}  // namespace nanquim::shaders\n")
file(WRITE ${GENERATED_DIRECTORY}/embedded_shaders.h.in "${EMBEDDED_SHADERS}")
configure_file(${GENERATED_DIRECTORY}/embedded_shaders.h.in ${GENERATED_DIRECTORY}/embedded_shaders.h COPYONLY)
